open Lean Elab Command

/-- Prints `#print axioms`-style lines for every theorem declared in the listed modules. -/
elab "#kernel_certificate_audit" : command => do
  let env ← getEnv
  let mods : Array Name := #[`«FrohmanianTether», `«FrohmanianTether».«AnalyticPipeline», `«FrohmanianTether».«ArnoldGeometric», `«FrohmanianTether».«AssemblyGap», `«FrohmanianTether».«Assumptions», `«FrohmanianTether».«ClayStatements», `«FrohmanianTether».«ClosureBlueprint», `«FrohmanianTether».«Combined», `«FrohmanianTether».«ConcreteSDiff», `«FrohmanianTether».«ConditionalRegularity», `«FrohmanianTether».«CubicSourceCertificate», `«FrohmanianTether».«ExplicitFlows», `«FrohmanianTether».«FeffermanClayA», `«FrohmanianTether».«ForMathlib».«Projection», `«FrohmanianTether».«HighSetDepletionAndBKM», `«FrohmanianTether».«IndependentMajorant», `«FrohmanianTether».«KineticEnergyDerivative», `«FrohmanianTether».«MathlibAgreement», `«FrohmanianTether».«NS_Equations», `«FrohmanianTether».«PhysicalSolutionSanity», `«FrohmanianTether».«Pillars», `«FrohmanianTether».«Pillars».«BKMContinuation», `«FrohmanianTether».«Pillars».«Uniqueness», `«FrohmanianTether».«PlanSteps», `«FrohmanianTether».«ScalingIdentities», `«FrohmanianTether».«StrainBoundRefutation», `«FrohmanianTether».«SymplecticConstraint», `«FrohmanianTether».«SymplecticTether», `«FrohmanianTether».«Takagi», `«FrohmanianTether».«TestFieldFamily», `«FrohmanianTether».«TetherC1», `«FrohmanianTether».«TetherDefinitionAudit», `«FrohmanianTether».«TetherEquivariantC1», `«FrohmanianTether».«TetherImpliesBKM», `«FrohmanianTether».«TetherNondegeneracy», `«FrohmanianTether».«TetherRateCounterexample», `«FrohmanianTether».«TetherRatePeriodicVacuity», `«FrohmanianTether».«TetherWeight», `«FrohmanianTether».«TetherWeightProof», `«FrohmanianTether».«TetheredLyapunov», `«FrohmanianTether».«TorusStrainExclusion», `«FrohmanianTether».«Uniqueness»]
  for m in mods do
    match env.getModuleIdx? m with
    | none => IO.println s!"MODULE NOT LOADED {m}"
    | some i =>
      for n in (env.header.moduleData[i.toNat]!).constNames do
        match env.find? n with
        | some (.thmInfo _) =>
          if !n.isInternal then
            let ax ← collectAxioms n
            let axs := ", ".intercalate (ax.toList.map toString)
            if ax.isEmpty then IO.println s!"'{n}' does not depend on any axioms [module {m}]"
            else IO.println s!"'{n}' depends on axioms: [{axs}] [module {m}]"
        | _ => pure ()

#kernel_certificate_audit
